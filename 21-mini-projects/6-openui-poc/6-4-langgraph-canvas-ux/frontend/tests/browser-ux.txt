async (page) => {
  await page.setViewportSize({width:1440,height:1000});
  await page.getByTestId('canvas-item').first().waitFor();
  const read = () => page.evaluate(() => JSON.parse(localStorage.getItem('canvas64:workspace:v2')));
  const results=[];
  const input = page.locator('[data-widget-id="sales"] input');
  await input.fill('새로고침 후에도 남는 입력');
  await input.blur();
  const before=await read();
  await page.getByRole('button',{name:'월별 매출 새로고침',exact:true}).click();
  await page.waitForFunction(old => JSON.parse(localStorage.getItem('canvas64:workspace:v2')).widgets.sales.snapshot.fetchedAt!==old, before.widgets.sales.snapshot.fetchedAt);
  if(await input.inputValue()!=='새로고침 후에도 남는 입력')throw new Error('Input reset during refresh');
  if((await read()).widgets.sales.program!==before.widgets.sales.program)throw new Error('Program changed on data refresh');
  results.push('refresh retains OpenUI program and live input');
  const handle=page.getByRole('button',{name:'월별 매출 크기 조절',exact:true});
  const box=await handle.boundingBox();await page.mouse.move(box.x+8,box.y+8);await page.mouse.down();await page.mouse.move(box.x+90,box.y+60,{steps:8});await page.mouse.up();
  if((await read()).widgets.sales.layout.w<=6)throw new Error('Pointer resize did not snap');
  results.push('pointer resize snaps to grid');
  await page.getByRole('button',{name:'새 Canvas',exact:true}).click();
  await page.locator('.canvas-tab.active button[title="Canvas 이름 변경"]').click();
  await page.getByRole('textbox',{name:'Canvas 이름',exact:true}).fill('보관함');
  await page.getByRole('button',{name:'Canvas 이름 저장',exact:true}).click();
  const target=(await read()).activeCanvasId;
  await page.getByRole('tab',{name:/내 대시보드/}).click();
  const card=page.locator('[data-widget-id="ads"]');
  await card.locator('.drag-handle').dragTo(page.getByRole('tab',{name:/보관함/}));
  if((await read()).widgets.ads.canvasId!==target)throw new Error('Cross-tab drag failed');
  await page.getByRole('tab',{name:/보관함/}).click();
  await page.getByTestId('chat-artifact').locator('.preview-drag').dragTo(page.getByTestId('canvas-drop-zone'));
  if((await read()).canvases.find(c=>c.id===target).widgetIds.length!==2)throw new Error('Chat drag copy failed');
  const copyId=(await read()).canvases.find(c=>c.id===target).widgetIds[1];
  const sourceInput=page.getByTestId('chat-artifact').locator('input');
  await sourceInput.fill('채팅 원본 수정');
  if((await read()).widgets[copyId].uiState['$memo']==='채팅 원본 수정')throw new Error('Chat and Canvas share state');
  results.push('create/rename/select tabs; cross-tab drag; chat drag creates independent copy');
  await page.locator(`[data-widget-id="${copyId}"] .drag-handle`).dragTo(page.locator('[data-widget-id="ads"] .card-toolbar'));
  if((await read()).canvases.find(c=>c.id===target).widgetIds[0]!==copyId)throw new Error('Card reorder failed');
  const tabs=page.locator('.canvas-tab');await tabs.nth(1).dragTo(tabs.nth(0));
  if((await read()).canvases[0].id!==target)throw new Error('Tab reorder failed');
  results.push('card and tab drag reorder');
  await page.reload();await page.getByTestId('canvas-item').first().waitFor();
  if((await read()).widgets[copyId].canvasId!==target)throw new Error('Layout restoration failed');
  await page.getByRole('tab',{name:/내 대시보드/}).click();
  if(await page.locator('[data-widget-id="sales"] input').inputValue()!=='새로고침 후에도 남는 입력')throw new Error('Persisted input did not restore');
  results.push('reload restores layout, tabs, charts and OpenUI inputs');
  await page.getByRole('tab',{name:/보관함/}).click();
  await page.getByRole('button',{name:'Canvas 비우기',exact:true}).click();
  await page.getByRole('button',{name:'취소',exact:true}).click();
  if((await read()).canvases[0].widgetIds.length!==2)throw new Error('Cancelled clear removed cards');
  await page.getByRole('button',{name:'Canvas 비우기',exact:true}).click();await page.getByRole('button',{name:'삭제',exact:true}).click();
  if((await read()).canvases[0].widgetIds.length)throw new Error('Clear failed');
  await page.locator('.canvas-tab.active button[title="Canvas 삭제"]').click();await page.getByRole('button',{name:'삭제',exact:true}).click();
  if((await read()).canvases.length!==1 || !(await read()).widgets.sales)throw new Error('Canvas delete damaged another canvas');
  results.push('clear/delete confirmations preserve other Canvas');
  await page.getByRole('button',{name:'테마 변경'}).click();await page.screenshot({path:'output/playwright/canvas-dark.png',fullPage:true});
  await page.setViewportSize({width:390,height:844});await page.screenshot({path:'output/playwright/canvas-mobile.png',fullPage:true});
  if(await page.evaluate(()=>document.documentElement.scrollWidth>window.innerWidth+1))throw new Error('Mobile horizontal overflow');
  await page.setViewportSize({width:1440,height:1000});
  return results;
}
